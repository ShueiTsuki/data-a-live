#if defined(PANDORA)
/*
 * Unattended runs on the 17 Pro (nobody to film the screen or to hold the power key):
 *
 * 1. Persistent console log in its own pinned file in userdata (ADT /chosen
 *    "pandora-log-extents" {file, partition, count} 4 KiB blocks + "pandora-log-blocks",
 *    made by tools/mkadt_pandora.py --log-extents from tools/pinfile's JSON). File block 0
 *    is a header, blocks 1..1024 hold sweet_log_ring (4 MiB):
 *      "PANDORLG" | u32 version 1 | u32 ring size | u64 total bytes | u64 flush seq
 *    ring offset of byte n = n % ring size (tools/pandora_log.py decodes it). Written
 *    every pandora_logsecs seconds (default 3) and once more from the restart hook, so a
 *    panic's text is kept. Needs ufsrw=1; fx_write_allowed accepts exactly these blocks.
 * 2. PE_halt_restart = PSCI SYSTEM_RESET: a panic (with debug= not containing DB_NMI 0x4),
 *    an iOS reboot, or boot-arg pandora_reboot_after=N seconds reset the phone, which
 *    then boots Android again.
 */
extern int (*PE_halt_restart)(unsigned int type);
#define PL_RING_BLOCKS  (SWEET_LOG_SIZE / UFS_BLOCK)

static bool pl_ready;
static uint64_t pl_flushed, pl_seq;
static int pl_chan;

static void
pl_psci_reset(void)
{
	register uint64_t x0 __asm__("x0") = 0x84000009ULL;     /* PSCI SYSTEM_RESET */
	__asm__ volatile("smc #0" : "+r"(x0) : : "x1", "x2", "x3", "memory");
	for (;;) {
		__builtin_arm_wfe();
	}
}

/* Write ring bytes not yet on disk, then the header. Caller owns the UFS host. */
static int
pl_flush(void)
{
	uint64_t total = os_atomic_load(&sweet_log_total, relaxed);
	if (total == pl_flushed) {
		return 0;
	}
	void *bounce = (void *)(su.dma + DMA_BOUNCE);
	uint64_t pos = total - pl_flushed >= SWEET_LOG_SIZE ? total - SWEET_LOG_SIZE : pl_flushed;
	int error = 0;
	while (pos < total && !error) {
		uint32_t off = (uint32_t)(pos % SWEET_LOG_SIZE);
		uint32_t blk = off / UFS_BLOCK;
		uint64_t need = ((uint64_t)off % UFS_BLOCK + (total - pos) + UFS_BLOCK - 1) / UFS_BLOCK;
		uint64_t lba, run;
		if (!pl_map(1 + blk, &lba, &run)) {
			return EINVAL;
		}
		uint32_t n = (uint32_t)MIN(MIN(MIN(need, (uint64_t)(PL_RING_BLOCKS - blk)), run),
		    (uint64_t)(BOUNCE_SIZE / UFS_BLOCK));
		memcpy(bounce, sweet_log_ring + (uint64_t)blk * UFS_BLOCK, (size_t)n * UFS_BLOCK);
		error = ufs_write_blocks(lba, n);
		pos += (uint64_t)(blk + n) * UFS_BLOCK - off;
	}
	if (!error) {
		struct sweet_log_header h = {
			.magic = { 'P', 'A', 'N', 'D', 'O', 'R', 'L', 'G' },
			.version = 1, .size = SWEET_LOG_SIZE, .total = total, .seq = ++pl_seq,
		};
		uint64_t lba, run;
		if (!pl_map(0, &lba, &run)) {
			return EINVAL;
		}
		bzero(bounce, UFS_BLOCK);
		memcpy(bounce, &h, sizeof(h));
		error = ufs_write_blocks(lba, 1);
	}
	if (!error) {
		error = ufs_sync_cache();
	}
	if (!error) {
		pl_flushed = total;
	}
	return error;
}

static int
pandora_halt_restart(unsigned int type)
{
	/* kPEHaltCPU 0, kPERestartCPU 1, kPEPanicRestartCPU 4, kPEPanicRestartCPUNoCallouts 9 */
	if (type != 0 && type != 1 && type != 4 && type != 9) {
		return 0;
	}
	if (pl_ready) {
		/* Everything else is stopped (panic) or going down: no lock. Let a command
		 * still in slot 0 finish first. */
		for (int i = 0; i < 100000 && (ufs_rd(REG_UTRLDBR) & 1); i++) {
			delay(1);
		}
		printf("PANDORA: restart type %u, final log flush %d\n", type, pl_flush());
		(void)pl_flush();
	}
	pl_psci_reset();
	return 0;
}

static void
pl_thread(__unused void *arg, __unused wait_result_t wr)
{
	uint32_t secs = 3, reboot_after = 0;
	PE_parse_boot_argn("pandora_logsecs", &secs, sizeof(secs));
	PE_parse_boot_argn("pandora_reboot_after", &reboot_after, sizeof(reboot_after));
	if (secs == 0) {
		secs = 3;
	}
	uint64_t start = mach_absolute_time(), limit = 0;
	if (reboot_after) {
		nanoseconds_to_absolutetime((uint64_t)reboot_after * NSEC_PER_SEC, &limit);
	}
	printf("PANDORA LOG: console log to the pinned log file every %u s%s%u\n", secs,
	    reboot_after ? ", reset after s " : "", reboot_after);
	for (;;) {
		(void)tsleep(&pl_chan, PZERO, "pandoralog", hz * (int)secs);
		lck_mtx_lock(&sweet_ufs_lock);
		int error = pl_flush();
		lck_mtx_unlock(&sweet_ufs_lock);
		if (error) {
			printf("PANDORA LOG: write failed (%d), log stopped\n", error);
			pl_ready = false;
			return;
		}
		if (limit && mach_absolute_time() - start >= limit) {
			printf("PANDORA: pandora_reboot_after=%u reached, resetting\n", reboot_after);
			lck_mtx_lock(&sweet_ufs_lock);
			(void)pl_flush();
			lck_mtx_unlock(&sweet_ufs_lock);
			pl_psci_reset();
		}
	}
}

static void
pandora_log_start(void)
{
	thread_t thread;
	PE_halt_restart = pandora_halt_restart;
	if (pl_next == 0 || !su.writable) {
		printf("PANDORA LOG: no log file (%u extents, writable %d); restart hook only\n",
		    pl_next, su.writable);
		return;
	}
	pl_ready = true;
	if (kernel_thread_start(pl_thread, NULL, &thread) == KERN_SUCCESS) {
		thread_deallocate(thread);
	}
}
#endif /* PANDORA */
